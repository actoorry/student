import psycopg2, pymysql, base64, requests, os

PG = { 'host':'101.126.146.118','port':5432,'dbname':'erp19c','user':'odoo','password':'odoo' }
MY = { 'host':'101.126.146.118','port':3306,'db':'suxin1','user':'root','passwd':'wf123456' }
API = 'http://localhost:48080/admin-api'  # 文件上传地址

def pg_fetch(sql):
    conn = psycopg2.connect(**PG)
    cur = conn.cursor()
    cur.execute(sql)
    rows = cur.fetchall()
    cols = [d[0] for d in cur.description]
    conn.close()
    return cols, rows

def my_exec(sql, params=None):
    conn = pymysql.connect(**MY)
    cur = conn.cursor()
    cur.execute(sql, params or ())
    conn.commit()
    conn.close()

def upload_image(datas_base64, filename):
    """解码 base64 并上传到 suxin 文件服务"""
    if not datas_base64:
        return None
    binary = base64.b64decode(datas_base64)
    # 保存为临时文件
    tmp = f'/tmp/{filename}'
    with open(tmp, 'wb') as f:
        f.write(binary)
    # 上传到 suxin
    with open(tmp, 'rb') as f:
        r = requests.post(f'{API}/infra/file/upload', files={'file': f})
    os.remove(tmp)
    if r.status_code == 200:
        return r.json().get('data', {}).get('url')
    return None
